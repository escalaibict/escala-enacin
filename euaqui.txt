    const loginScreen = document.getElementById('login-screen');
    const formScreen = document.getElementById('form-screen');
    const adminScreen = document.getElementById('admin-screen');
    const userInfo = document.getElementById('user-info');

        loginScreen.classList.add("hidden");

        if (user.email === admEmail) {

    function login() {
      const email = document.getElementById("email").value;
      const password = document.getElementById("password").value;
      auth.signInWithEmailAndPassword(email, password).catch(alert);
    }

    function signup() {
      const email = document.getElementById("email").value;
      const password = document.getElementById("password").value;
      auth.createUserWithEmailAndPassword(email, password).catch(alert);
    }

    function logout() {
      auth.signOut();
    }

    function enviarEscala() {
      const user = auth.currentUser;
      if (!user) return;

      const escala = {
        uid: user.uid,
        email: user.email,
        nome: document.getElementById("nome").value,
        dia1: document.getElementById("dia1").value,
        dia2: document.getElementById("dia2").value,
        dia3: document.getElementById("dia3").value,
        observacoes: document.getElementById("observacoes").value,
        timestamp: new Date()
      };

      db.collection("escalas").add(escala)
        .then(() => {
          alert("Escala enviada!");
          document.getElementById("form-screen").reset();
        })
        .catch(err => alert("Erro: " + err));
    }

    function carregarRegistros() {
      db.collection("escalas").orderBy("timestamp", "desc").onSnapshot(snapshot => {
        const container = document.getElementById("registros");
        container.innerHTML = "";
        snapshot.forEach(doc => {
          const data = doc.data();
          container.innerHTML += `
            <div class="record">
              <strong>${data.nome}</strong> (${data.email})<br>
              Dia 1: ${data.dia1} | Dia 2: ${data.dia2} | Dia 3: ${data.dia3}<br>
              Obs: ${data.observacoes || "Nenhuma"}<br>
              <small>${new Date(data.timestamp.toDate()).toLocaleString()}</small>
            </div>
          `;
        });
      });
    }
