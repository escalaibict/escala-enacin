  <div class="container">

    <!-- Tela de login/cadastro -->

      <label>Email:</label>
      <input type="email" id="email">
      <label>Senha:</label>
      <input type="password" id="password">
      <button onclick="login()">Entrar</button>
      <button onclick="signup()">Cadastrar</button>
    </div>

    <!-- Formulário da Escala -->
    <div id="formSection" class="hidden">
      <h3>Preencher Escala</h3>
      <label>Nome:</label>
      <input type="text" id="nome">
      <label>Telefone:</label>
      <input type="tel" id="telefone">
      
      <div id="escalaCampos"></div>
      <button onclick="enviarEscala()">Enviar Escala</button>
      <button onclick="logout()">Sair</button>
    </div>

    <!-- Área ADM -->
    <div id="adminSection" class="hidden">
      <h3>Escalas Preenchidas</h3>
      <table id="adminTable">
        <thead>
          <tr id="tableHeader"></tr>
        </thead>
        <tbody id="tableBody"></tbody>
      </table>

  <script>
    // Firebase config (substitua depois)
    const firebaseConfig = {
      apiKey: "SUA_API_KEY",
      authDomain: "SEU_AUTH_DOMAIN",
      projectId: "SEU_PROJECT_ID",
      storageBucket: "SEU_BUCKET",
      messagingSenderId: "SEU_SENDER_ID",
      appId: "SEU_APP_ID"
    };
    firebase.initializeApp(firebaseConfig);
    const auth = firebase.auth();
    const db = firebase.firestore();

    const datasEscala = [
      "20/05", "21/05", "22/05", "23/05", "24/05",
      "27/05", "28/05", "29/05", "30/05", "31/05"
    ];
    const opcoes = ["Dia inteiro", "Meio turno", "Ausente"];

    const escalaCampos = document.getElementById('escalaCampos');
    datasEscala.forEach(data => {
      const label = document.createElement('label');
      label.innerText = `Dia ${data}:`;
      const select = document.createElement('select');
      select.id = `data-${data}`;
      opcoes.forEach(op => {
        const option = document.createElement('option');
        option.value = op;
        option.innerText = op;
        select.appendChild(option);
      });
      escalaCampos.appendChild(label);
      escalaCampos.appendChild(select);
    });

    function login() {
      const email = document.getElementById('email').value;
      const senha = document.getElementById('password').value;
      auth.signInWithEmailAndPassword(email, senha)
        .then(() => location.reload())
        .catch(e => alert(e.message));
    }

    function signup() {
      const email = document.getElementById('email').value;
      const senha = document.getElementById('password').value;
      auth.createUserWithEmailAndPassword(email, senha)
        .then(() => location.reload())
        .catch(e => alert(e.message));
    }

    function logout() {
      auth.signOut().then(() => location.reload());
    }

    auth.onAuthStateChanged(user => {
      if (user) {
        const email = user.email;
        document.getElementById('authSection').classList.add('hidden');

        if (email === 'admin@admin.com') {
          document.getElementById('adminSection').classList.remove('hidden');
          carregarTabelaADM();
        } else {
          document.getElementById('formSection').classList.remove('hidden');
        }
      }
    });

    function enviarEscala() {
      const user = auth.currentUser;
      const nome = document.getElementById('nome').value;
      const telefone = document.getElementById('telefone').value;

      const dados = {
        nome,
        email: user.email,
        telefone,
        dataEnvio: new Date().toISOString()
      };

      datasEscala.forEach(data => {
        const valor = document.getElementById(`data-${data}`).value;
        dados[data] = valor;
      });

      db.collection("escalas").add(dados).then(() => {
        alert("Escala enviada com sucesso!");
      }).catch(e => {
        alert("Erro ao enviar: " + e.message);
      });
    }

    function carregarTabelaADM() {
      const header = document.getElementById('tableHeader');
      const body = document.getElementById('tableBody');

      header.innerHTML = `<th>Nome</th><th>Email</th><th>Telefone</th>`;
      datasEscala.forEach(data => {
        const th = document.createElement('th');
        th.innerText = data;
        header.appendChild(th);
      });

      db.collection("escalas").get().then(snapshot => {
        body.innerHTML = "";
        snapshot.forEach(doc => {
          const dados = doc.data();
          const tr = document.createElement('tr');
          tr.innerHTML = `
            <td>${dados.nome}</td>
            <td>${dados.email}</td>
            <td>${dados.telefone}</td>
          `;
          datasEscala.forEach(data => {
            tr.innerHTML += `<td>${dados[data] || '-'}</td>`;
          });
          body.appendChild(tr);
        });
      });
    }
  </script>
