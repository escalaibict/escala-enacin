    function showLogin() {
      document.getElementById("register-container").classList.add("hidden");
      document.getElementById("login-container").classList.remove("hidden");
    }

    function showRegister() {
      document.getElementById("login-container").classList.add("hidden");
      document.getElementById("register-container").classList.remove("hidden");
    }

    function login() {
      const email = document.getElementById("email").value;
      const password = document.getElementById("password").value;
      auth.signInWithEmailAndPassword(email, password)
        .then((userCredential) => {
          if (email === adminEmail) {
            showAdmin();
          } else {
            document.getElementById("login-container").classList.add("hidden");
            document.getElementById("form-container").classList.remove("hidden");
          }
        })
        .catch((error) => {
          document.getElementById("login-error").innerText = error.message;
        });
    }

    function register() {
      const email = document.getElementById("reg-email").value;
      const password = document.getElementById("reg-password").value;
      auth.createUserWithEmailAndPassword(email, password)
        .then(() => showLogin())
        .catch((error) => {
          document.getElementById("register-error").innerText = error.message;
        });
    }

    function logout() {
      auth.signOut().then(() => {
        location.reload();
      });
    }

    function enviarEscala() {
      const user = auth.currentUser;
      if (!user) return;
      const escala = {
        nome: document.getElementById("nome").value,
        email: user.email,
        dia1: document.getElementById("dia1").value,
        dia2: document.getElementById("dia2").value,
        dia3: document.getElementById("dia3").value,
        dia4: document.getElementById("dia4").value,
        timestamp: firebase.firestore.FieldValue.serverTimestamp()
      };
      db.collection("escalas").add(escala).then(() => {
        alert("Escala enviada!");
        logout();
      });
    }

    function showAdmin() {
      document.getElementById("login-container").classList.add("hidden");
      document.getElementById("admin-container").classList.remove("hidden");
      const tbody = document.querySelector("#escala-tabela tbody");
      tbody.innerHTML = "";
      db.collection("escalas").orderBy("timestamp", "desc").get().then(snapshot => {
        snapshot.forEach(doc => {
          const data = doc.data();
          const tr = document.createElement("tr");
          tr.innerHTML = `
            <td>${data.nome}</td>
            <td>${data.email}</td>
            <td>${data.dia1}</td>
            <td>${data.dia2}</td>
            <td>${data.dia3}</td>
            <td>${data.dia4}</td>
          `;
          tbody.appendChild(tr);
        });
      });
    }
