  <!-- Firebase -->
  <script type="module">
    // CONFIGURAÇÃO FIREBASE (Substitua pelos seus dados do Firebase)
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.11.1/firebase-app.js";
    import {
      getAuth,
      onAuthStateChanged,
      signInWithEmailAndPassword,
      createUserWithEmailAndPassword,
      signOut,
    } from "https://www.gstatic.com/firebasejs/10.11.1/firebase-auth.js";
    import {
      getFirestore,
      collection,
      addDoc,
      getDocs,
      query,
      where,
      serverTimestamp
    } from "https://www.gstatic.com/firebasejs/10.11.1/firebase-firestore.js";

    const firebaseConfig = {
      apiKey: "SUA_API_KEY",
      authDomain: "SEU_AUTH_DOMAIN",
      projectId: "SEU_PROJECT_ID",
      storageBucket: "SEU_STORAGE_BUCKET",
      messagingSenderId: "SEU_SENDER_ID",
      appId: "SEU_APP_ID"
    };

    const app = initializeApp(firebaseConfig);
    const auth = getAuth(app);
    const db = getFirestore(app);

    const loginDiv = document.getElementById("loginDiv");
    const formDiv = document.getElementById("formDiv");
    const tabelaDiv = document.getElementById("tabelaDiv");
    const logoutBtn = document.getElementById("logoutBtn");

    const ADMIN_EMAIL = "escaladepresencaibict@gmail.com";

    function login() {
      const email = document.getElementById("email").value;
      const password = document.getElementById("password").value;
      signInWithEmailAndPassword(auth, email, password)
        .then(() => {})
        .catch((error) => {
          document.getElementById("loginMessage").innerText = "Erro ao fazer login: " + error.message;
        });
    }

    function signup() {
      const email = document.getElementById("email").value;
      const password = document.getElementById("password").value;
      createUserWithEmailAndPassword(auth, email, password)
        .then(() => {})
        .catch((error) => {
          document.getElementById("loginMessage").innerText = "Erro ao cadastrar: " + error.message;
        });
    }

    function logout() {
      signOut(auth);
    }

    async function enviarEscala() {
      const nome = document.getElementById("nome").value;
      const lotacao = document.getElementById("lotacao").value;
      const subunidade = document.getElementById("subunidade").value;
      const gestor = document.getElementById("gestor").value;
      const mes = document.getElementById("mesReferencia").value;
      const tipo = document.getElementById("tipoEscala").value;
      const obs = document.getElementById("observacoes").value;

      const user = auth.currentUser;

      if (!nome || !lotacao || !gestor || !mes || !tipo) {
        alert("Preencha todos os campos obrigatórios.");
        return;
      }

      await addDoc(collection(db, "escalas"), {
        uid: user.uid,
        email: user.email,
        nome,
        lotacao,
        subunidade,
        gestor,
        mes,
        tipo,
        observacoes: obs,
        timestamp: serverTimestamp()
      });

      alert("Escala enviada com sucesso!");
      document.getElementById("formDiv").reset();
    }

    async function carregarEscalas(usuario) {
      let q = collection(db, "escalas");
      if (usuario.email !== ADMIN_EMAIL) {
        q = query(q, where("uid", "==", usuario.uid));
      }
      const snap = await getDocs(q);
      let html = "<table><tr><th>Nome</th><th>Lotação</th><th>Subunidade</th><th>Gestor</th><th>Mês</th><th>Tipo</th><th>Observações</th></tr>";
      snap.forEach(doc => {
        const data = doc.data();
        html += `<tr>
          <td>${data.nome}</td>
          <td>${data.lotacao}</td>
          <td>${data.subunidade}</td>
          <td>${data.gestor}</td>
          <td>${data.mes}</td>
          <td>${data.tipo}</td>
          <td>${data.observacoes || ''}</td>
        </tr>`;
      });
      html += "</table>";
      document.getElementById("tabelaContainer").innerHTML = html;
    }

    onAuthStateChanged(auth, (user) => {
      if (user) {
        loginDiv.classList.add("hidden");
        formDiv.classList.remove("hidden");
        tabelaDiv.classList.remove("hidden");
        logoutBtn.classList.remove("hidden");
        carregarEscalas(user);
      } else {
        loginDiv.classList.remove("hidden");
        formDiv.classList.add("hidden");
        tabelaDiv.classList.add("hidden");
        logoutBtn.classList.add("hidden");
      }
    });
  </script>
